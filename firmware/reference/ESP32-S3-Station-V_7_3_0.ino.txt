// ฟังก์ชันร่วมกับ Host: เวลาที่ไม่มีหยดแล้วถือว่า "หยุดไหล" (ต้องใช้สูตรเดียวกัน)

uint32_t calcOcclusionTimeoutMs(float rateHr, uint8_t df) {
  uint32_t timeoutMs = 8000;
  if (rateHr > 0 && df > 0) {
    float gttPerMin = rateHr * df / 60.0f;
    if (gttPerMin > 0) {
      uint32_t d = (uint32_t)((60000.0f / gttPerMin) * 2.5f);
      if (d > timeoutMs) timeoutMs = d;
    }
  }
  if (timeoutMs > 60000) timeoutMs = 60000;
  return timeoutMs;
}

bool isLocalOcclusion(unsigned long now) {
  return isRunning && hasFirstDropOccurred &&
         (now - lastDropTimestamp > calcOcclusionTimeoutMs(targetRateHr, dropFactor));
}

// รหัสเตือนที่ใช้งานจริงบนเครื่องนี้: ตรวจเองได้ (หยุดไหล) + รับจาก Host (เร็ว/ช้า/หมด)
uint8_t getEffectiveAlert(unsigned long now) {
  if (!isRunning) return ALERT_NONE;
  if (isHostOnline && currentAlertState == ALERT_BAG_EMPTY) return ALERT_BAG_EMPTY;
  if (isLocalOcclusion(now)) return ALERT_OCCLUSION;
  if (isHostOnline && (currentAlertState == ALERT_TOO_FAST || currentAlertState == ALERT_TOO_SLOW ||
                       currentAlertState == ALERT_OCCLUSION)) return currentAlertState;
  return ALERT_NONE;
}

// เติมช่องว่างซ้าย-ขวาให้ข้อความยาวคงที่ เพื่อเขียนทับของเดิมได้โดยไม่ต้องล้างพื้นที่ (ไม่กะพริบ)
String padCenter(const String &s, unsigned int width) {
  if (s.length() >= width) return s;
  unsigned int total = width - s.length();
  unsigned int left = total / 2;
  String out;
  out.reserve(width);
  for (unsigned int i = 0; i < left; i++) out += ' ';
  out += s;
  for (unsigned int i = 0; i < total - left; i++) out += ' ';
  return out;
}

void handlePassiveBuzzerEngine() {
  unsigned long now = millis();
  if (buzzerBeepUntil > 0 && now >= buzzerBeepUntil) {
    noTone(BUZZER_PIN);
    buzzerBeepUntil = 0;
  }

  if (now < snoozeUntilTime) return;
  if (currentState == STATE_CONFIG_ID || isHoldUiActive) return;
  if (now < buzzerNextBeepTime) return;

  switch (getEffectiveAlert(now)) {
    case ALERT_OCCLUSION: beepNonBlocking(1800, 150); buzzerNextBeepTime = now + 900;  break;
    case ALERT_TOO_FAST:  beepNonBlocking(2200, 80);  buzzerNextBeepTime = now + 700;  break;
    case ALERT_TOO_SLOW:  beepNonBlocking(1200, 120); buzzerNextBeepTime = now + 1500; break;
    case ALERT_BAG_EMPTY: beepNonBlocking(1500, 250); buzzerNextBeepTime = now + 1500; break;
    default: break;
  }
}

int sampleSensorAveragePrecise(int samples = 40) {
  long sum = 0;
  for (int i = 0; i < samples; i++) {
    sum += analogRead(SENSOR_AO_PIN);
    delayMicroseconds(500);
  }
  return (int)(sum / samples);
}

int waitAndMonitorSensorStep(const char* stepTitle, const char* stepDescTh, const char* stepDescEn, uint16_t boxColor) {
  noTone(BUZZER_PIN);
  buzzerBeepUntil = 0;

  tft.fillRoundRect(6, 36, 160, 160, 8, THEME_DARKBOX);
  tft.drawRoundRect(6, 36, 160, 160, 8, boxColor);
  drawCenteredString(stepTitle, 44, 1, COLOR_YELLOW, THEME_DARKBOX);
  drawCenteredString(stepDescEn, 60, 2, COLOR_WHITE, THEME_DARKBOX);
  drawCenteredString(stepDescTh, 82, 1, THEME_PINK, THEME_DARKBOX);

  tft.fillRoundRect(20, 102, 132, 44, 6, THEME_NAVY);
  tft.drawRoundRect(20, 102, 132, 44, 6, boxColor);
  drawCenteredString("SENSOR LIVE ADC", 106, 1, THEME_SKYBLUE, THEME_NAVY);
  drawCenteredString("Click Button >>", 176, 1, COLOR_WHITE, THEME_DARKBOX);

  unsigned long lastAdcUpdate = 0;
  int lastVal = -1;
  while (digitalRead(BTN_PIN) == HIGH) {
    if (millis() - lastAdcUpdate >= 60) {
      lastAdcUpdate = millis();
      int liveAdc = analogRead(SENSOR_AO_PIN);
      if (liveAdc != lastVal) {
        lastVal = liveAdc;
        tft.fillRect(24, 120, 124, 20, THEME_NAVY);
        drawCenteredString(String(liveAdc), 120, 2, COLOR_WHITE, THEME_NAVY);
      }
    }
    delay(5);
  }

  modalSafeBeep(2400, 40);
  while (digitalRead(BTN_PIN) == LOW) delay(10);
  delay(60);

  tft.fillRect(20, 102, 132, 44, THEME_NAVY);
  drawCenteredString("SAMPLING...", 118, 1, COLOR_YELLOW, THEME_NAVY);
  int avgVal = sampleSensorAveragePrecise(50);
  modalSafeBeep(1800, 60);

  return avgVal;
}

void executeButtonCalibrationWizard() {
  noTone(BUZZER_PIN);
  buzzerBeepUntil = 0;

  tft.fillScreen(THEME_BG);
  tft.fillRect(0, 0, 172, 28, THEME_PINK);
  drawCenteredString("CALIBRATION WIZARD", 8, 1, COLOR_WHITE, THEME_PINK);

  int newLiquid = waitAndMonitorSensorStep("STEP 1 OF 2", "(กระเปาะมีสารน้ำ)", "FILL LIQUID", THEME_SKYBLUE);
  int newAir    = waitAndMonitorSensorStep("STEP 2 OF 2", "(กระเปาะเปล่าไม่มีน้ำ)", "EMPTY CHAMBER", THEME_PINK);

  int diff = abs(newLiquid - newAir);
  tft.fillRoundRect(6, 204, 160, 108, 8, THEME_NAVY);
  tft.drawRoundRect(6, 204, 160, 108, 8, (diff >= 80) ? COLOR_GREEN : COLOR_RED);

  if (diff >= 80) {
    valLiquid     = newLiquid;
    valAir        = newAir;
    dropThreshold = (valLiquid + valAir) / 2;
    dropLowersAdc = (valLiquid < valAir);   // รองรับเซนเซอร์ทั้งสองขั้ว
    hysteresis    = (int)(diff * 0.20);
    if (hysteresis < 60)  hysteresis = 60;
    if (hysteresis > 300) hysteresis = 300;

    stationPrefs.begin("st_cal", false);
    stationPrefs.putInt("v_liq", valLiquid);
    stationPrefs.putInt("v_air", valAir);
    stationPrefs.putInt("th", dropThreshold);
    stationPrefs.putInt("hys", hysteresis);
    stationPrefs.end();

    drawCenteredString("CALIBRATION OK!", 212, 1, COLOR_GREEN, THEME_NAVY);
    drawCenteredString(dropLowersAdc ? "Mode: drop = LOW ADC" : "Mode: drop = HIGH ADC", 232, 1, COLOR_WHITE, THEME_NAVY);
    modalSafeBeep(2000, 100);
    delay(100);
    modalSafeBeep(2500, 150);
    delay(1500);
  } else {
    // ไม่บันทึกค่าที่ล้มเหลว — คงค่าเดิมไว้
    drawCenteredString("FAILED: DIFF LOW!", 216, 1, COLOR_RED, THEME_NAVY);
    drawCenteredString("Keep previous values", 234, 1, COLOR_WHITE, THEME_NAVY);
    modalSafeBeep(800, 200);
    delay(1500);
  }
  dropState = false;
  lastUserActivityTime = millis();
  stateNeedsRedraw = true;
}

#if defined(ESP_IDF_VERSION) && ESP_IDF_VERSION >= ESP_IDF_VERSION_VAL(5, 0, 0)
void OnTimeSyncRecv(const esp_now_recv_info_t *info, const uint8_t *incomingDataBytes, int len) {
  int rssi = (info && info->rx_ctrl) ? info->rx_ctrl->rssi : -55;
#else
void OnTimeSyncRecv(const uint8_t *mac, const uint8_t *incomingDataBytes, int len) {
  int rssi = -55;
#endif
  // ทำงานใน WiFi task: คัดลอกข้อมูลแล้วส่งต่อให้ loop() เท่านั้น
  if (len != sizeof(struct_host_sync)) return;   // แพ็กเก็ตของ Station อื่น (23 bytes) ถูกกรองออกที่นี่
  struct_host_sync syncData;
  memcpy(&syncData, incomingDataBytes, sizeof(syncData));
  if (syncData.protocolVersion != PROTOCOL_VERSION) return;

  portENTER_CRITICAL(&syncMux);
  lastHostSeenMs = millis();
  pendingHostRssi = rssi;
  if (syncData.stationId == currentStationId) {
    pendingOwnSync = syncData;
    pendingOwnSyncAvailable = true;
  }
  portEXIT_CRITICAL(&syncMux);
}

void resetDropCounters() {
  totalDrops = 0;
  periodDropsCounter = 0;
  currentFlowRate_ml_hr = 0.0;
  currentGttMin = 0.0;
  totalVolumeMl = 0.0;
  hasFirstDropOccurred = false;
  rateAnchorValid = false;
  dropState = false;
  snoozeUntilTime = 0;
  lastDrawnDrops = 0xFFFFFFFF;
  stateNeedsRedraw = true;
  beepNonBlocking(2400, 120);
}

void processHostLink(unsigned long now) {
  struct_host_sync s;
  bool haveOwn = false;
  unsigned long seen;
  int rssi;

  portENTER_CRITICAL(&syncMux);
  seen = lastHostSeenMs;
  rssi = pendingHostRssi;
  if (pendingOwnSyncAvailable) {
    s = pendingOwnSync;
    pendingOwnSyncAvailable = false;
    haveOwn = true;
  }
  portEXIT_CRITICAL(&syncMux);
  now = millis();   // อ่านหลังคัดลอก เพื่อไม่ให้ now < seen

  bool online = (seen != 0) && (now - seen <= HOST_TIMEOUT_MS);
  if (online) {
    lastHostRssi = rssi;
    if (channelScanning) {
      channelScanning = false;
      stationPrefs.begin("st_cfg", false);
      stationPrefs.putUChar("ch", espnowChannel);
      stationPrefs.end();
    }
  } else if (isHostOnline) {
    hostOfflineSince = now;
    currentAlertState = ALERT_NONE;   // ค่าจาก Host เก่าเกินไป ไม่ใช้ต่อ (ยังตรวจหยุดไหลเองได้)
  }
  isHostOnline = online;

  if (haveOwn) {
    if (s.epochTime > 1600000000UL) {
      struct timeval tv = { (time_t)s.epochTime, 0 };
      settimeofday(&tv, NULL);
      isClockSynced = (s.isSynced == 1);
    }
    targetRateHr      = s.targetRateHr;
    totalPlanMl       = s.totalPlanMl;
    currentAlertState = s.alertCode;
    if (s.dropFactor == 10 || s.dropFactor == 15 || s.dropFactor == 20 || s.dropFactor == 60) {
      dropFactor = s.dropFactor;
    }
    if (!resetSeqKnown) {
      lastResetSeq = s.resetSeq;
      resetSeqKnown = true;
    } else if (s.resetSeq != lastResetSeq) {
      lastResetSeq = s.resetSeq;
      resetDropCounters();
    }
  }

  // ค้นหาช่องสัญญาณของ Host เมื่อขาดการติดต่อเกิน Timeout
  if (!isHostOnline && (now - hostOfflineSince >= HOST_TIMEOUT_MS) &&
      (now - lastChannelHopTime >= CHANNEL_SCAN_DWELL_MS)) {
    espnowChannel = (espnowChannel % 13) + 1;
    esp_wifi_set_channel(espnowChannel, WIFI_SECOND_CHAN_NONE);
    lastChannelHopTime = now;
    channelScanning = true;
  }
}

// ตรวจจับหยดน้ำอย่างเงียบสนิท
void checkAnalogDrop() {
  unsigned long now = millis();

  if (isRunning) {
    int sensorValue = analogRead(SENSOR_AO_PIN);
    bool dropDetected, dropReleased;
    if (dropLowersAdc) {
      dropDetected = sensorValue < dropThreshold;
      dropReleased = sensorValue > (dropThreshold + hysteresis);
    } else {
      dropDetected = sensorValue > dropThreshold;
      dropReleased = sensorValue < (dropThreshold - hysteresis);
    }

    if (!dropState && dropDetected) {
      if (now - lastDebounceTime > minDropInterval) {
        dropState = true;
        totalDrops++;
        periodDropsCounter++;
        lastDebounceTime = now;

        isDropFalling = true;
        dropFallStartTime = now;
        animDropY = CHAMBER_NOZZLE_Y;

        if (rateAnchorValid) {
          unsigned long dropDeltaMs = now - lastDropTimestamp;
          if (dropDeltaMs > minDropInterval && dropDeltaMs < 60000) {
            float instantGttMin = 60000.0f / (float)dropDeltaMs;
            float instantRateMlHr = (instantGttMin * 60.0f) / (float)dropFactor;
            if (currentFlowRate_ml_hr == 0.0f) currentFlowRate_ml_hr = instantRateMlHr;
            else currentFlowRate_ml_hr = (instantRateMlHr * 0.75f) + (currentFlowRate_ml_hr * 0.25f);
          }
        }
        rateAnchorValid = true;
        hasFirstDropOccurred = true;
        lastDropTimestamp = now;
      }
    }
    else if (dropState && dropReleased) {
      dropState = false;
    }
  }

  // อัตราไหลลดลงตามเวลาจริงเมื่อหยดช้ากว่าที่คาด และเป็น 0 เมื่อเกิน Occlusion timeout
  if (isRunning && hasFirstDropOccurred && currentFlowRate_ml_hr > 0.0f) {
    unsigned long elapsed = now - lastDropTimestamp;
    float expectedMs = 60000.0f / (currentFlowRate_ml_hr * dropFactor / 60.0f);
    if (elapsed > expectedMs * 1.5f && elapsed > 0) {
      float capRate = (60000.0f / (float)elapsed) * 60.0f / (float)dropFactor;
      if (capRate < currentFlowRate_ml_hr) currentFlowRate_ml_hr = capRate;
    }
    if (elapsed > calcOcclusionTimeoutMs(targetRateHr, dropFactor)) currentFlowRate_ml_hr = 0.0f;
  }
  currentGttMin = (currentFlowRate_ml_hr * dropFactor) / 60.0f;
  totalVolumeMl = totalDrops / (float)dropFactor;
}

String getStationClockStr() {
  time_t now;
  time(&now);
  struct tm *timeinfo = localtime(&now);
  char buf[12];
  if (isClockSynced) strftime(buf, sizeof(buf), "%H:%M", timeinfo);
  else snprintf(buf, sizeof(buf), "--:--");
  return String(buf);
}

void drawSignalBarsRaw(int x, int y, int rssi, bool online, int bars, uint16_t barColor) {

  int heights[4] = {4, 7, 10, 13};
  for (int i = 0; i < 4; i++) {
    int bx = x + (i * 5);
    int bh = heights[i];
    int by = y + (13 - bh);
    tft.fillRect(bx, by, 3, bh, (i < bars) ? barColor : THEME_DARKBOX);

  }

int rssiToBars(int rssi, bool online, uint16_t &barColor) {
  barColor = COLOR_GREEN;
  if (!online) return 0;
  if (rssi >= -65) { barColor = COLOR_GREEN;  return 4; }
  if (rssi >= -75) { barColor = THEME_CYAN;   return 3; }
  if (rssi >= -85) { barColor = COLOR_ORANGE; return 2; }
  barColor = COLOR_RED;
  return 1;
}

// วาดเฉพาะเมื่อสถานะเปลี่ยน / ขณะหยดกำลังตก (จำกัด ~60 fps) เพื่อไม่ให้ SPI แย่งเวลาการอ่านเซนเซอร์
void updateDripAnimation() {
  if (currentState != STATE_NORMAL_VIEW || currentNursePage != 1 || isHoldUiActive) return;

  unsigned long now = millis();
  int mode;   // 0 = รอหยด, 1 = หยดกำลังตก, 2 = พักการนับ, 3 = ไม่มีการไหล
  if (!isRunning) mode = 2;
  else if (isDropFalling) mode = 1;
  else if (isLocalOcclusion(now)) mode = 3;
  else mode = 0;

  if (mode != animMode) {
    bool fallToIdle = (animMode == 1 && mode == 0);
    bool idleToFall = (animMode == 0 && mode == 1);
    if (fallToIdle) {
      tft.fillCircle(138, animDropPrevY, 2, THEME_DARKBOX);
      tft.fillRoundRect(124, 122, 28, 12, 2, THEME_CYAN);   // กระเพื่อมที่ผิวน้ำ
      poolFlashUntil = now + 120;
      tft.fillCircle(138, CHAMBER_NOZZLE_Y, 2, THEME_CYAN);
    } else if (!idleToFall) {
      drawChamberStatic();
      if (mode == 2 || mode == 3) {
        tft.setTextColor(COLOR_YELLOW, THEME_DARKBOX);
        tft.setTextSize(2);
        tft.setCursor(mode == 2 ? 128 : 134, 96);
        tft.print(mode == 2 ? "||" : "!");
      } else if (mode == 0) {
        tft.fillCircle(138, CHAMBER_NOZZLE_Y, 2, THEME_CYAN);
      }
    }
    animDropPrevY = CHAMBER_NOZZLE_Y;
    animMode = mode;
  }

  if (mode == 1) {
    static unsigned long lastFrame = 0;
    if (now - lastFrame < 16) return;
    lastFrame = now;
    unsigned long elapsed = now - dropFallStartTime;
    if (elapsed <= DROP_FALL_DURATION) {
      animDropY = CHAMBER_NOZZLE_Y + (int)((float)elapsed / (float)DROP_FALL_DURATION * (CHAMBER_POOL_Y - CHAMBER_NOZZLE_Y));
      if (animDropY != animDropPrevY) {
        tft.fillCircle(138, animDropPrevY, 2, THEME_DARKBOX);
        tft.fillCircle(138, animDropY, 2, THEME_CYAN);
        animDropPrevY = animDropY;
      }
    } else {
      isDropFalling = false;
    }
  } else if (mode == 0 && poolFlashUntil != 0 && now >= poolFlashUntil) {
    tft.fillRoundRect(124, 122, 28, 12, 2, THEME_NAVY);
    poolFlashUntil = 0;
  }
}

void handleButton() {
  static unsigned long btnPressStart = 0;
  static bool isHolding = false;
  static int clickCount = 0;
  static unsigned long lastReleaseTime = 0;

  bool reading = (digitalRead(BTN_PIN) == LOW);
  unsigned long now = millis();

  if (reading && !isHolding) {
    btnPressStart = now;
    isHolding = true;
    isHoldUiActive = false;
    holdBeep3sDone = false;
    holdBeep5sDone = false;
    lastUserActivityTime = now;
  }
  else if (reading && isHolding) {
    unsigned long holdDur = now - btnPressStart;
    if (holdDur >= 400 && currentState != STATE_CONFIG_ID) {
      isHoldUiActive = true;
      if (holdDur >= 3000 && !holdBeep3sDone) {
        beepNonBlocking(2200, 50);
        holdBeep3sDone = true;
      }
      if (holdDur >= 5000 && !holdBeep5sDone) {
        beepNonBlocking(2600, 80);
        holdBeep5sDone = true;
      }
      if (now - lastHoldRenderTime >= 35) {
        lastHoldRenderTime = now;
        drawHoldProgressHUD(holdDur);
      }
    }
  }
  else if (!reading && isHolding) {
    unsigned long pressDur = now - btnPressStart;
    isHolding = false;
    lastUserActivityTime = now;

    if (isHoldUiActive) {
      isHoldUiActive = false;
      if (pressDur >= 5000) {
        soundScreensaver();
        currentState = STATE_CREDIT;
        stateNeedsRedraw = true;
        clickCount = 0;
        return;
      }
      else if (pressDur >= 3000) {
        executeButtonCalibrationWizard();
        if (currentState == STATE_SCREENSAVER) currentState = STATE_NORMAL_VIEW;
        clickCount = 0;
        return;
      }
      else {
        soundClick();
        stateNeedsRedraw = true;
        clickCount = 0;
        return;
      }
    }
    else if (pressDur > 30 && pressDur < 400) {
      clickCount++;
      lastReleaseTime = now;
    }
  }

  if (clickCount > 0 && !isHolding && (now - lastReleaseTime > 280)) {
    if (currentState == STATE_CONFIG_ID) {
      soundClick();
      tempConfigStationId = (tempConfigStationId % 8) + 1;
      configAutoSaveTimeout = now + 4000;
      clickCount = 0;
      return;
    }

    if (currentState == STATE_EMERGENCY) {
      snoozeUntilTime = now + SNOOZE_DURATION_MS;
      noTone(BUZZER_PIN);
      buzzerBeepUntil = 0;
      soundClick();
      currentState = STATE_NORMAL_VIEW;
      stateNeedsRedraw = true;
      clickCount = 0;
      return;
    }

    if (currentState == STATE_CREDIT || currentState == STATE_SCREENSAVER) {
      // คลิกกี่ครั้งก็ได้ = กลับหน้าหลัก (ไม่เปลี่ยนสถานะการนับโดยไม่ตั้งใจ)
      soundClick();
      currentState = STATE_NORMAL_VIEW;
      stateNeedsRedraw = true;
      clickCount = 0;
      return;
    }

    if (clickCount >= 3) {
      soundClick();
      noTone(BUZZER_PIN);
      setLedColor(50, 0, 50);
      lastLedColorCode = -1;
      tempConfigStationId = currentStationId;
      configAutoSaveTimeout = now + 4000;
      currentState = STATE_CONFIG_ID;
      stateNeedsRedraw = true;
    }
    else if (clickCount == 2) {
      isRunning = !isRunning;
      currentFlowRate_ml_hr = 0.0f;
      dropState = false;
      if (isRunning) {
        // เริ่มนับต่อ: ไม่ใช้ช่วงเวลาที่พักไว้คำนวณอัตรา และเริ่มจับเวลาหยุดไหลใหม่
        rateAnchorValid = false;
        lastDropTimestamp = now;
        soundStart();
      } else {
        soundStop();
      }
      stateNeedsRedraw = true;
    }
    else if (clickCount == 1) {
      currentNursePage = (currentNursePage % 3) + 1;
      stateNeedsRedraw = true;
      soundClick();
    }
    clickCount = 0;
  }
}

void updateStatusLed(unsigned long now) {
  int code;
  if (currentState == STATE_CONFIG_ID) return;
  if (currentState == STATE_EMERGENCY) code = 1;
  else if (getEffectiveAlert(now) != ALERT_NONE) code = 2;   // มีเหตุแต่พักเสียงอยู่
  else if (!isRunning) code = 3;
  else if (!isHostOnline) code = 4;
  else code = 5;
  if (code == lastLedColorCode) return;
  lastLedColorCode = code;
  switch (code) {
    case 1: setLedColor(90, 0, 0);  break;   // แดง: ฉุกเฉิน
    case 2: setLedColor(60, 20, 0); break;   // ส้ม: พักเสียงเตือน
    case 3: setLedColor(50, 40, 0); break;   // เหลือง: พักการนับ
    case 4: setLedColor(0, 0, 60);  break;   // น้ำเงิน: ไม่พบ Host
    default: setLedColor(0, 40, 0); break;   // เขียว: ปกติ
  }
}

void setup() {
  Serial.begin(115200);

  // กำหนดเขตเวลา ICT UTC+7 ประเทศไทยถาวร
  setenv("TZ", "ICT-7", 1);
  tzset();

  pinMode(BUZZER_PIN, OUTPUT); noTone(BUZZER_PIN);
  pinMode(TFT_BLK, OUTPUT); digitalWrite(TFT_BLK, HIGH);
  pinMode(BTN_PIN, INPUT_PULLUP);
  pinMode(SENSOR_AO_PIN, INPUT);

  analogReadResolution(12);
  analogSetAttenuation(ADC_11db);

  SPI_TFT.begin(TFT_SCLK, -1, TFT_MOSI, TFT_CS);
  tft.init(172, 320);
  tft.setSPISpeed(40000000);
  tft.setRotation(0);
  tft.setTextWrap(false);

  stationPrefs.begin("st_cfg", false);
  currentStationId = stationPrefs.getUChar("id", 1);
  espnowChannel    = stationPrefs.getUChar("ch", DEFAULT_ESPNOW_CHANNEL);
  stationPrefs.end();
  if (currentStationId < 1 || currentStationId > 8) currentStationId = 1;
  if (espnowChannel < 1 || espnowChannel > 13) espnowChannel = DEFAULT_ESPNOW_CHANNEL;

  stationPrefs.begin("st_cal", true);
  valLiquid     = stationPrefs.getInt("v_liq", 1200);
  valAir        = stationPrefs.getInt("v_air", 2800);
  dropThreshold = stationPrefs.getInt("th", 2000);
  hysteresis    = stationPrefs.getInt("hys", 200);
  stationPrefs.end();
  dropLowersAdc = (valLiquid < valAir);

  setLedColor(0, 40, 80);
  tft.fillScreen(THEME_BG);
  tft.drawRoundRect(4, 4, 164, 312, 8, THEME_SKYBLUE);
  tft.drawRoundRect(6, 6, 160, 308, 6, THEME_PINK);

  drawCenteredString("SMART IV", 35, 3, THEME_SKYBLUE, THEME_BG);
  drawCenteredString("ALERT OS", 65, 2, THEME_PINK, THEME_BG);

  tft.fillRoundRect(16, 120, 140, 50, 6, THEME_PINK);
  drawCenteredString("BED " + String(currentStationId < 10 ? "0" : "") + String(currentStationId), 134, 3, COLOR_WHITE, THEME_PINK);

  drawCenteredString("Clinical Trial Node", 200, 1, COLOR_WHITE, THEME_BG);
  drawCenteredString("BCN Phrae Innovation", 220, 1, THEME_CYAN, THEME_BG);
  drawCenteredString("Firmware v" + String(APP_VERSION), 240, 1, THEME_SKYBLUE, THEME_BG);

  playWelcomeMelody();
  delay(600);

  WiFi.mode(WIFI_STA);
  WiFi.disconnect();
  WiFi.setSleep(false);
  esp_wifi_set_channel(espnowChannel, WIFI_SECOND_CHAN_NONE);

  if (esp_now_init() == ESP_OK) {
    esp_now_register_recv_cb(esp_now_recv_cb_t(OnTimeSyncRecv));

    esp_now_peer_info_t peerInfo = {};
    memcpy(peerInfo.peer_addr, broadcastAddress, 6);
    peerInfo.channel = 0;          // 0 = ช่องปัจจุบัน (รองรับการเปลี่ยนช่องขณะค้นหา Host)
    peerInfo.ifidx = WIFI_IF_STA;
    peerInfo.encrypt = false;
    esp_now_add_peer(&peerInfo);
  } else {
    drawCenteredString("ESP-NOW INIT FAILED", 270, 1, COLOR_RED, THEME_BG);
    delay(2000);
  }

  unsigned long now = millis();
  hostOfflineSince = now;
  lastChannelHopTime = now;
  lastUserActivityTime = now;

  currentState = STATE_NORMAL_VIEW;
  currentNursePage = 1;
  stateNeedsRedraw = true;
}

void loop() {
  checkAnalogDrop();
  handleButton();
  handlePassiveBuzzerEngine();

  unsigned long currentMillis = millis();
  processHostLink(currentMillis);

  if (currentState == STATE_CONFIG_ID) {
    if (stateNeedsRedraw) drawStationIdConfigFramework();
    static unsigned long lastConfigDraw = 0;
    if (currentMillis - lastConfigDraw >= 250) {
      lastConfigDraw = currentMillis;
      updateStationIdConfigDynamic();
    }

    if (currentMillis >= configAutoSaveTimeout) {
      if (tempConfigStationId != currentStationId) {
        currentStationId = tempConfigStationId;
        stationPrefs.begin("st_cfg", false);
        stationPrefs.putUChar("id", currentStationId);
        stationPrefs.end();
        // เปลี่ยนเตียง: ล้างค่าที่ได้จาก Host ของเตียงเดิม รอค่าของเตียงใหม่
        targetRateHr = 0; totalPlanMl = 0; currentAlertState = ALERT_NONE;
        resetSeqKnown = false;
      }
      soundStart();
      lastLedColorCode = -1;
      lastUserActivityTime = currentMillis;
      currentState = STATE_NORMAL_VIEW;
      stateNeedsRedraw = true;
    }
  } else {
    uint8_t effectiveAlert = getEffectiveAlert(currentMillis);
    bool isEmergencyCondition = (effectiveAlert != ALERT_NONE);

    if (isEmergencyCondition && currentMillis >= snoozeUntilTime) {
      if (!isHoldUiActive) {
        if (currentState != STATE_EMERGENCY || effectiveAlert != lastEmergencyAlert) {
          currentState = STATE_EMERGENCY;
          stateNeedsRedraw = true;
        }
      }
    } else if (!isEmergencyCondition && currentState == STATE_EMERGENCY) {
      currentState = STATE_NORMAL_VIEW;
      stateNeedsRedraw = true;
    }
    if (!isEmergencyCondition) snoozeUntilTime = 0;

    // Screensaver อัตโนมัติเมื่อไม่มีการใช้งาน
    if (currentState == STATE_NORMAL_VIEW && !isHoldUiActive &&
        currentMillis - lastUserActivityTime >= SCREENSAVER_IDLE_MS) {
      currentState = STATE_SCREENSAVER;
      stateNeedsRedraw = true;
    }

    if (!isHoldUiActive) {
      if (currentState == STATE_EMERGENCY) {
        if (stateNeedsRedraw) drawEmergencyScreen();
      }
      else if (currentState == STATE_NORMAL_VIEW) {
        if (currentNursePage == 1) {
          if (stateNeedsRedraw) { drawPage1Framework(); updatePage1Dynamic(); }
          updateDripAnimation();
        } else if (currentNursePage == 2) {
          if (stateNeedsRedraw) { drawPage2Framework(); updatePage2Dynamic(); }
        } else if (currentNursePage == 3) {
          if (stateNeedsRedraw) { drawPage3Framework(); updatePage3Dynamic(); }
        }
      }
      else if (currentState == STATE_SCREENSAVER) {
        if (stateNeedsRedraw) { drawScreensaverFramework(); updateScreensaverDynamic(); }
      }
      else if (currentState == STATE_CREDIT) {
        if (stateNeedsRedraw) drawDeveloperCreditScreen();
      }
    }

    static unsigned long lastDynamicUpdate = 0;
    if (currentMillis - lastDynamicUpdate >= 1000) {
      lastDynamicUpdate = currentMillis;

      if (!isHoldUiActive) {
        if (currentState == STATE_NORMAL_VIEW) {
          if (currentNursePage == 1) updatePage1Dynamic();
          else if (currentNursePage == 2) updatePage2Dynamic();
          else if (currentNursePage == 3) updatePage3Dynamic();
        } else if (currentState == STATE_SCREENSAVER) {
          updateScreensaverDynamic();
        } else if (currentState == STATE_EMERGENCY) {
          drawEmergencyRateValues();
        }
      }
    }

    updateStatusLed(currentMillis);
  }

  // ส่งสถานะให้ Host ทุก 1 วินาที (ส่งต่อแม้อยู่ในหน้าตั้งค่า เพื่อไม่ให้ Host เห็นเป็น Offline)
  if (currentMillis - lastSendTime >= STATUS_SEND_INTERVAL_MS) {
    unsigned long now = millis();
    myData.protocolVersion = PROTOCOL_VERSION;
    myData.stationId       = currentStationId;
    myData.isRunning       = isRunning ? 1 : 0;
    myData.totalDrops      = totalDrops;
    myData.periodDrops     = periodDropsCounter;
    myData.flowRateHr      = isRunning ? currentFlowRate_ml_hr : 0.0f;
    myData.msSinceLastDrop = hasFirstDropOccurred ? (uint32_t)(now - lastDropTimestamp) : 0;
    myData.batteryVolts    = readStationBattery();

    if (esp_now_send(broadcastAddress, (uint8_t *)&myData, sizeof(myData)) == ESP_OK) {
      periodDropsCounter = 0;
    }
    lastSendTime = currentMillis;
  }
}
